cmake_minimum_required(VERSION 3.20)
project(solid_modeling_e1 LANGUAGES CXX)
option(E1_FREESTANDING "Use installed Windows Clang/LLD without a C runtime" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(E1_FREESTANDING)
  if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Freestanding build requires native Windows Clang")
  endif()
  get_filename_component(LLVM_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
  add_library(e1_core OBJECT src/evaluator.cpp src/timing.cpp)
  target_include_directories(e1_core PRIVATE include)
  target_compile_definitions(e1_core PRIVATE E1_FREESTANDING=1)
  target_compile_options(e1_core PRIVATE -O3 -ffreestanding -fno-exceptions -fno-rtti -fno-stack-protector -fno-fast-math -ffp-contract=off -fno-math-errno -Wall -Wextra)
  set(E1_LIBRARY "${CMAKE_BINARY_DIR}/e1_native.dll")
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/e1_kernel32.lib"
    COMMAND "${LLVM_BIN}/lld.exe" -flavor link /lib /machine:x64 "/def:${CMAKE_CURRENT_SOURCE_DIR}/toolchain/kernel32.def" "/out:${CMAKE_BINARY_DIR}/e1_kernel32.lib"
    DEPENDS toolchain/kernel32.def VERBATIM)
  add_custom_command(OUTPUT "${E1_LIBRARY}"
    COMMAND "${LLVM_BIN}/lld.exe" -flavor link /dll /noentry /nodefaultlib "/out:${E1_LIBRARY}" $<TARGET_OBJECTS:e1_core> "${CMAKE_BINARY_DIR}/e1_kernel32.lib"
    DEPENDS e1_core $<TARGET_OBJECTS:e1_core> "${CMAKE_BINARY_DIR}/e1_kernel32.lib" COMMAND_EXPAND_LISTS VERBATIM)
  add_custom_target(e1_native ALL DEPENDS "${E1_LIBRARY}")
else()
  add_library(e1_native SHARED src/evaluator.cpp src/timing.cpp)
  target_include_directories(e1_native PUBLIC include)
  if(MSVC)
    target_compile_options(e1_native PRIVATE /O2 /fp:strict /W4)
  else()
    target_compile_options(e1_native PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra)
  endif()
  set(E1_LIBRARY "$<TARGET_FILE:e1_native>")
endif()
enable_testing()
set(E1_PYTHON "" CACHE FILEPATH "Existing reference Python interpreter for parity tests")
if(E1_PYTHON)
  add_test(NAME e1_numeric_parity COMMAND "${E1_PYTHON}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native.py" --library "${E1_LIBRARY}")
  set_tests_properties(e1_numeric_parity PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
endif()
